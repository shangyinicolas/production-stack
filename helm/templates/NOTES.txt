{{- if .Values.routerSpec.enableRouter }}
{{- $svc := printf "%s-router-service" .Release.Name }}
{{- $port := .Values.routerSpec.servicePort }}

Send requests to the router:
{{- if and .Values.routerSpec.ingress.enabled .Values.routerSpec.ingress.hosts }}
{{- $scheme := ternary "https" "http" (not (empty .Values.routerSpec.ingress.tls)) }}
  export ROUTER_URL={{ $scheme }}://{{ (first .Values.routerSpec.ingress.hosts).host }}
{{- else if eq .Values.routerSpec.serviceType "NodePort" }}
  export NODE_PORT=$(kubectl get svc {{ $svc }} -n {{ .Release.Namespace }} -o jsonpath='{.spec.ports[0].nodePort}')
  export NODE_IP=$(kubectl get nodes -o jsonpath='{.items[0].status.addresses[0].address}')
  export ROUTER_URL=http://$NODE_IP:$NODE_PORT
{{- else if eq .Values.routerSpec.serviceType "LoadBalancer" }}
  The external IP can take a minute to be assigned:
  export ROUTER_IP=$(kubectl get svc {{ $svc }} -n {{ .Release.Namespace }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
  export ROUTER_URL=http://$ROUTER_IP:{{ $port }}
{{- else }}
  kubectl port-forward svc/{{ $svc }} -n {{ .Release.Namespace }} 30080:{{ $port }}
  export ROUTER_URL=http://localhost:30080
{{- end }}

List the served models:
  curl $ROUTER_URL/v1/models
{{- with (first $models) }}

Try a completion:
  curl $ROUTER_URL/v1/completions -H "Content-Type: application/json" \
    -d '{"model": "{{ .modelURL }}", "prompt": "Hello, my name is", "max_tokens": 16}'
{{- end }}
